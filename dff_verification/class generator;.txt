class generator;
  transaction tr;
  mailbox #(transaction) mbx;
  int count;
  function new(mailbox #(transaction) mbx);
    this.mbx=mbx;
  endfunction
  task run();
    repeat(count)
      begin
        tr=new();
        assert(tr.randomize());
        mbx.put(tr);
         $display("[%4t] [GEN] Generated : reset=%0b d=%0b", $time, tr.reset, tr.d);
      end
  endtask
endclass
